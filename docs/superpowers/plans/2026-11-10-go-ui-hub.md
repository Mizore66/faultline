# UI Hub and Turn Dashboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port the frozen `ui` hub and discovery, with verified Git/turn artifact routes, a dedicated Go turn dashboard, and raw JSON inspection.

**Architecture:** Add filesystem discovery and hub routes on the existing Go proofview transport. Keep Git-proof rendering unchanged; verify turn bundles with `turnproof.Verify`, then project `investigation.json` into an escaped read-only dashboard while retaining raw JSON. CLI reuses the existing opener and `awaitBundleInterrupt`.

**Tech Stack:** Go 1.27 standard library, existing `jsjson`/`nodefs`/`turnproof`, pinned Node 22.22.2 as oracle only.

**Spec:** `docs/superpowers/specs/2026-11-10-go-ui-hub-design.md`

## Global Constraints

- Work on a dedicated UI branch; no protected-source edit, workflow edit, automatic merge, release, oracle retirement, or evidence deletion.

- Preserve `serve --bundle`, plain `serve`, native execution/verifier rules, export-only behavior, and old golden/corpus bytes.
- Turn bundles are verified before dashboard rendering. No unverified dashboard, mutation route, new 409, blanket normalization, or production entropy override.

func turnDashboardFixture(t *testing.T) jsjson.Value {
    t.Helper()
    str := jsjson.MakeString
    state := func(id string, ordinal int, verdict string) jsjson.Value {
        return jsjson.MustObject(
            "turnId", str(id), "turnOrdinal", jsjson.MakeNumber(float64(ordinal)),
            "verdict", str(verdict), "treeDigest", str("tree-"+id), "snapshotDigest", str("snapshot-"+id),
            "executionIds", jsjson.MakeArray(str("exec-"+id)), "runIds", jsjson.MakeArray(str("run-"+id)))
    }
    return jsjson.MustObject(
        "status", str("COMPLETED"),
        "proof", jsjson.MustObject(
            "isProof", jsjson.MakeBool(true), "reason", str("<reason>"),
            "evidenceGrade", str("TURN_PATH_EVIDENCE_GRADE_EXPERIMENTAL"), "evidenceLabel", str("label")),
        "environmentHomogeneity", str("HOMOGENEOUS"),
        "introduction", jsjson.MustObject("status", str("ATTRIBUTED"), "reason", str("<intro>")),
        "errors", jsjson.MakeArray(str("example error")),
        "stableStates", jsjson.MakeArray(state("turn-1", 1, "PASS"), state("turn-2", 2, "FAIL")),
        "transitions", jsjson.MakeArray(jsjson.MustObject(
            "kind", str("PASS_TO_FAIL"),
            "before", jsjson.MustObject("turnOrdinal", jsjson.MakeNumber(1)),
            "after", jsjson.MustObject("turnOrdinal", jsjson.MakeNumber(2)))),
        "repository", str("repo"), "ledgerDigest", str("ledger"), "executionsPerState", jsjson.MakeNumber(3),
        "recorder", str("codex-turn-tree-replay"), "nativeCodexInterception", jsjson.MakeBool(false),
        "runtimeMapping", jsjson.MustObject())
}

func TestRenderTurnDashboardContract(t *testing.T) {
    result := turnDashboardFixture(t)
    got := RenderTurnDashboard(result)
    for _, text := range []string{"COMPLETED", "PROOF", "TURN_PATH_EVIDENCE_GRADE_EXPERIMENTAL", "turn-1", "turn-2", "PASS_TO_FAIL", "run-1", "run-2", "tree-1", "snapshot-1"} {
        if !strings.Contains(got, text) { t.Errorf("dashboard lacks %q", text) }
    }
    if strings.Contains(got, "<script>") { t.Fatal("unescaped dynamic script") }
    if !strings.Contains(got, "&lt;reason&gt;") { t.Fatal("reason not escaped") }
}
